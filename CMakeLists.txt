cmake_minimum_required(VERSION 3.25)
if(DEFINED CMAKE_CXX_COMPILER)
  get_filename_component(REBORN_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
  if(EXISTS "${REBORN_COMPILER_DIR}/x86_64-w64-mingw32-windres.exe")
    if(NOT IS_ABSOLUTE "${CMAKE_RC_COMPILER}" OR NOT EXISTS "${CMAKE_RC_COMPILER}")
      set(CMAKE_RC_COMPILER "${REBORN_COMPILER_DIR}/x86_64-w64-mingw32-windres.exe" CACHE FILEPATH "Windows resource compiler" FORCE)
    endif()
  endif()
endif()
if(DEFINED CMAKE_RC_COMPILER)
  file(TO_CMAKE_PATH "${CMAKE_RC_COMPILER}" REBORN_RC_PATH)
  set(CMAKE_RC_COMPILER "${REBORN_RC_PATH}" CACHE FILEPATH "Windows resource compiler" FORCE)
endif()
project(JStarsReborn VERSION 0.4.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This runtime requires native Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(REBORN_ASAN "Enable AddressSanitizer with MSVC" OFF)
add_library(reborn_options INTERFACE)
target_compile_definitions(reborn_options INTERFACE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(reborn_options INTERFACE /W4 /permissive- /EHsc)
  if(REBORN_ASAN)
    target_compile_options(reborn_options INTERFACE /fsanitize=address /Zi)
    target_link_options(reborn_options INTERFACE /INCREMENTAL:NO)
  endif()
else()
  target_compile_options(reborn_options INTERFACE -Wall -Wextra -Wpedantic)
  if(REBORN_ASAN)
    message(FATAL_ERROR "This ASan configuration is only supported with MSVC; no unverified MinGW sanitizer claim")
  endif()
endif()
if(MINGW)
  target_link_options(reborn_options INTERFACE -static)
endif()
add_library(reborn_paths src/core/paths.cpp)
target_include_directories(reborn_paths PUBLIC include)
target_link_libraries(reborn_paths PUBLIC reborn_options shell32 ole32)
add_library(reborn_core src/core/core.cpp src/core/crash.cpp)
target_include_directories(reborn_core PUBLIC include)
target_link_libraries(reborn_core PUBLIC reborn_options reborn_paths user32)
add_library(reborn_rendering src/rendering/renderer.cpp)
target_include_directories(reborn_rendering PUBLIC include)
target_link_libraries(reborn_rendering PUBLIC reborn_options d3d12 dxgi dxguid)
add_library(reborn_input src/input/input.cpp)
target_include_directories(reborn_input PUBLIC include)
target_link_libraries(reborn_input PUBLIC reborn_options xinput9_1_0)
add_library(reborn_audio src/audio/audio.cpp)
target_include_directories(reborn_audio PUBLIC include)
target_link_libraries(reborn_audio PUBLIC reborn_options xaudio2_9 ole32)
# 0.2.x native foundation (DX12/XAudio2/XInput smoke application). Not part of the 0.4.0 release.
add_executable(JStarsFoundation WIN32 src/platform/main.cpp)
target_link_libraries(JStarsFoundation PRIVATE reborn_core reborn_rendering reborn_input reborn_audio shell32 user32)
if(MINGW)
  target_link_options(JStarsFoundation PRIVATE -municode)
endif()

enable_language(RC)
# Platform-independent, read-only game-dump validation (also built under sanitizers by tools/portable_tests.sh).
add_library(reborn_gamecheck src/setup/sha256.cpp src/setup/game_validate.cpp)
target_include_directories(reborn_gamecheck PUBLIC include)
target_link_libraries(reborn_gamecheck PUBLIC reborn_options)
add_library(reborn_setup src/setup/inventory.cpp src/setup/install.cpp)
target_include_directories(reborn_setup PUBLIC include)
target_link_libraries(reborn_setup PUBLIC reborn_options reborn_gamecheck reborn_paths bcrypt)
add_executable(JStarsSetup WIN32 src/setup/main.cpp resources/setup.rc)
target_include_directories(JStarsSetup PRIVATE resources)
target_link_libraries(JStarsSetup PRIVATE reborn_setup comctl32 ole32 shell32 user32 gdi32 uuid)
if(MINGW)
  target_link_options(JStarsSetup PRIVATE -municode)
endif()

if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE reborn_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(input_tests tests/input_tests.cpp)
  target_link_libraries(input_tests PRIVATE reborn_input)
  add_test(NAME input COMMAND input_tests)
  add_executable(paths_tests tests/paths_tests.cpp)
  target_link_libraries(paths_tests PRIVATE reborn_paths)
  add_test(NAME paths COMMAND paths_tests)
  add_executable(gamecheck_tests tests/gamecheck_tests.cpp)
  target_link_libraries(gamecheck_tests PRIVATE reborn_gamecheck)
  # Cross-check the portable SHA-256 against the independently recorded digest of the
  # shipped runtime when both are present in the source tree.
  set(REBORN_SELFCHECK_ARGS)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/setup-payload.txt" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/JStarsReborn.exe")
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/setup-payload.txt" reborn_runtime_line REGEX "\tJStarsReborn\.exe$")
    if(reborn_runtime_line)
      list(GET reborn_runtime_line 0 reborn_runtime_line)
      string(SUBSTRING "${reborn_runtime_line}" 0 64 reborn_runtime_sha)
      set(REBORN_SELFCHECK_ARGS "${CMAKE_CURRENT_SOURCE_DIR}/JStarsReborn.exe" "${reborn_runtime_sha}")
    endif()
  endif()
  add_test(NAME gamecheck COMMAND gamecheck_tests ${REBORN_SELFCHECK_ARGS})
  foreach(check setup_inventory setup_install pkg_intake)
    add_executable(${check}_tests tests/${check}_tests.cpp)
    target_link_libraries(${check}_tests PRIVATE reborn_setup)
    add_test(NAME ${check} COMMAND ${check}_tests)
  endforeach()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME asset_audit COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_asset_audit.py -v)
    set_tests_properties(asset_audit PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()
# 0.4.0 launcher: JStarsReborn.exe (starts runtime\jstars-runtime.exe, built from runtime/ and gpu/; see docs/BUILD_RELEASE.md).
add_library(reborn_prepare src/prepare/prepare.cpp)
target_include_directories(reborn_prepare PUBLIC include src/prepare)
target_link_libraries(reborn_prepare PUBLIC reborn_options reborn_gamecheck)
add_executable(JStarsReborn WIN32 src/launcher/main.cpp src/launcher/launcher_core.cpp resources/launcher.rc)
target_include_directories(JStarsReborn PRIVATE src/launcher resources)
target_link_libraries(JStarsReborn PRIVATE reborn_prepare reborn_paths comctl32 ole32 oleaut32 shell32 shlwapi uuid windowscodecs msimg32 gdi32 user32)
if(MINGW)
  target_link_options(JStarsReborn PRIVATE -municode)
endif()

install(TARGETS JStarsReborn JStarsSetup RUNTIME DESTINATION .)
install(DIRECTORY config/ DESTINATION config)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES README.md CHANGELOG.md milestones.json THIRD_PARTY_NOTICES.md PROJECT_AUDIT.md supported-versions.tsv DESTINATION .)
install(PROGRAMS tools/asset_audit.py tools/smoke_test.py tools/make_setup_payload.py DESTINATION tools)

install(DIRECTORY licenses/ DESTINATION licenses)
install(DIRECTORY evidence/ DESTINATION evidence)
install(DIRECTORY assets/ DESTINATION assets)
install(DIRECTORY mods/ DESTINATION mods)
install(FILES toolchain-lock.json DESTINATION .)
